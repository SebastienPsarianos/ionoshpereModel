\documentclass[border=10pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\IfFileExists{../macros.tex}{\input{../macros}}{\input{macros}}

\begin{document}
\begin{tikzpicture}[>=latex]


	% --- Parameters ---
	\def\dtheta{2.0}
	\def\Nlon{8}
	\def\maxR{4.0}         % 2 * dtheta

	% =============================================
	%  MAIN DIAGRAM
	% =============================================


	% --- Concentric colatitude rings ---
	\draw[ringcol,thick] (0,0) circle ({\dtheta});
	\draw[ringcol,thick] (0,0) circle ({2*\dtheta});

	% --- Longitude lines ---
	\foreach \j in {0,...,\the\numexpr\Nlon-1\relax} {
			\pgfmathsetmacro{\angle}{\j * 360 / \Nlon}
			\draw[thick, ringcol] (0,0) -- (\angle:\maxR + \dtheta / 2);
		}

	% --- Grid dots: first ring
	\foreach \j in {0,...,\the\numexpr\Nlon-1\relax} {
			\pgfmathsetmacro{\angle}{\j * 360 / \Nlon}
			\fill[adjcol] (\angle:{\dtheta}) circle (3pt);
		}

	% --- Grid dots: second ring ---
	\foreach \j in {0,...,\the\numexpr\Nlon-1\relax} {
			\pgfmathsetmacro{\angle}{\j * 360 / \Nlon}
			\fill[gridcol] (\angle:{2*\dtheta}) circle (2pt);
		}

	% --- Pole point (on top of everything) ---
	\fill[polecol] (0,0) circle (3.5pt);

	% =============================================
	%  RADIUS LABELS (above horizontal line)
	% =============================================
	\node[anchor=south west,  font=\large, annotcol, inner sep=2pt]
	at ({\dtheta}, 0.1) {$\Delta\theta$};
	\node[anchor=south west, font=\large, annotcol, inner sep=2pt]
	at ({2*\dtheta}, 0.1) {$2\Delta\theta$};


	% =============================================
	%  LONGITUDE SPACING (arc at outer edge)
	% =============================================

	\draw[thick, <->, annotcol]
	(90:{\maxR + 0.4}) arc (90:{90 + 360/\Nlon}:{\maxR + 0.4})
	node[midway, above, font=\small] {$\Delta\phi$};

	% =============================================
	%  LEGEND (to the right)
	% =============================================

	\begin{scope}[shift={(\maxR + 2.8, 1.625)}]

		\node[anchor=north, font=\small\bfseries, annotcol] at (1.9, -0.2) {Legend};

		% Pole node
		\fill[polecol] (0.35, -1.2) circle (3.5pt);
		\node[anchor=west, font=\footnotesize, annotcol] at (0.5, -1.2)
		{Pole node};

		% Adjacent nodes
		\fill[adjcol] (0.35, -1.75) circle (3pt);
		\node[anchor=west, font=\footnotesize, annotcol] at (0.5, -1.75)
		{Adjacent nodes};

		% Grid nodes
		\fill[gridcol] (0.35, -2.3) circle (2pt);
		\node[anchor=west, font=\footnotesize, annotcol] at (0.5, -2.3)
		{Grid nodes};

		% Colatitude ring
		\draw[ringcol, thick] (0.2, -2.85) -- (0.5, -2.85);
		\node[anchor=west, font=\footnotesize, annotcol] at (0.5, -2.85)
		{Grid};

		% Bounding box
		\draw[annotcol!40, rounded corners=3pt]
		(0, 0) rectangle (3.8, -3.25);

	\end{scope}

\end{tikzpicture}
\end{document}
